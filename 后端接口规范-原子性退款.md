**后端实现要点**：

```java
@PostMapping("/contract/{contractId}/atomic-refund")
@Transactional  // 🔑 必须添加事务注解
public ResponseEntity<?> atomicRefund(
    @PathVariable Long contractId,
    @RequestBody RefundRequest request) {
  
  try {
    // 1️⃣ 查询合同信息
    Contract contract = contractService.getContract(contractId);
    if (contract == null) throw new RuntimeException("合同不存在");
    
    Long tenantId = contract.getTenantId();
    Long landlordId = contract.getLandlordId();
    BigDecimal refundAmount = request.getRefundAmount();
    
    // 2️⃣ 查询账户（加锁防并发）
    Wallet tenantWallet = walletService.getWalletWithLock(tenantId);
    Wallet landlordWallet = walletService.getWalletWithLock(landlordId);
    
    // 3️⃣ 记录操作前状态（用于验证）
    BigDecimal beforeSum = 
        tenantWallet.getBalance()
        .add(landlordWallet.getBalance());
    
    // 4️⃣ 执行原子性更新
    // 租客余额 += refundAmount
    tenantWallet.setBalance(
        tenantWallet.getBalance().add(refundAmount)
    );
    walletService.updateWallet(tenantWallet);
    
    // 房东余额 -= refundAmount
    landlordWallet.setBalance(
        landlordWallet.getBalance().subtract(refundAmount)
    );
    walletService.updateWallet(landlordWallet);
    
    // 5️⃣ 记录交易日志
    Transaction txn = new Transaction();
    txn.setTransactionType("REFUND");
    txn.setAmount(refundAmount);
    txn.setFromUserId(landlordId);
    txn.setToUserId(tenantId);
    txn.setRelatedNo(contract.getContractNo());
    transactionService.saveTransaction(txn);
    
    // 6️⃣ 验证资金守恒（重要！）
    BigDecimal afterSum = 
        tenantWallet.getBalance()
        .add(landlordWallet.getBalance());
    
    if (!beforeSum.equals(afterSum)) {
        throw new RuntimeException("资金守恒检查失败");
    }
    
    // 7️⃣ 更新合同状态
    contract.setStatus("TERMINATED");
    contractService.updateContract(contract);
    
    // 8️⃣ 返回成功
    return ResponseEntity.ok(new ApiResponse(
        0, "退款成功", 
        new RefundResponse(
            txn.getId(),
            refundAmount,
            tenantWallet.getBalance(),
            landlordWallet.getBalance(),
            afterSum
        )
    ));
    
  } catch (Exception e) {
    // ❌ 异常会自动触发 @Transactional 回滚
    log.error("退款失败，操作已回滚", e);
    
    return ResponseEntity
        .status(409)
        .body(new ApiResponse(409, "资金守恒检查失败，操作已回滚", null));
  }
}
```

## 🛡️ 数据库锁机制

### 防止并发操作导致的资金错乱

```sql
-- 1️⃣ 行级锁（SELECT FOR UPDATE）
SELECT * FROM wallet 
WHERE user_id = ? 
FOR UPDATE;  -- 获得排他锁，其他事务必须等待

-- 2️⃣ 事务内操作（原子性保证）
START TRANSACTION;
  UPDATE wallet SET balance = balance - 3000 
  WHERE user_id = 2;  -- 租客扣款
  
  UPDATE wallet SET balance = balance + 3000 
  WHERE user_id = 1;  -- 房东入款
  
  INSERT INTO transaction_log (...);
COMMIT;  -- 所有变化一起提交或全部回滚

-- 3️⃣ 并发测试场景
-- 连接 A: 支付 1000 元
-- 连接 B: 同时退款 500 元
-- 预期结果：两个操作都成功或都失败，不能部分成功
```

## ✅ 资金守恒验证算法

```javascript
/**
 * 验证资金守恒（服务端）
 * 在事务提交前后分别执行
 */
function validateFundsConservation(
    beforeTenantBalance,
    beforeLandlordBalance,
    afterTenantBalance,
    afterLandlordBalance,
    transactionAmount
) {
    // 计算变化前后的总额
    const beforeSum = 
        new Decimal(beforeTenantBalance)
        .plus(new Decimal(beforeLandlordBalance));
    
    const afterSum = 
        new Decimal(afterTenantBalance)
        .plus(new Decimal(afterLandlordBalance));
    
    // 验证守恒（允许 0.01 元浮点误差）
    const tolerance = new Decimal('0.01');
    const difference = beforeSum.minus(afterSum).abs();
    
    if (difference.greaterThan(tolerance)) {
        throw new ConservationViolationException(
            `资金守恒检查失败: ` +
            `操作前总额=${beforeSum}, ` +
            `操作后总额=${afterSum}, ` +
            `差异=${difference}`
        );
    }
    
    return {
        isValid: true,
        beforeSum: beforeSum.toNumber(),
        afterSum: afterSum.toNumber(),
        difference: difference.toNumber()
    };
}
```

## 📊 数据库设计

### Wallet 表结构

```sql
CREATE TABLE wallet (
  user_id BIGINT PRIMARY KEY,
  available_balance DECIMAL(12, 2) NOT NULL DEFAULT 0,
  frozen_amount DECIMAL(12, 2) NOT NULL DEFAULT 0,
  total_income DECIMAL(15, 2) NOT NULL DEFAULT 0,
  total_expense DECIMAL(15, 2) NOT NULL DEFAULT 0,
  last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  version INT DEFAULT 0,  -- 用于乐观锁
  UNIQUE KEY uk_user (user_id)
);

-- 索引优化
CREATE INDEX idx_user_balance ON wallet(user_id, available_balance);
```

### Transaction 表结构

```sql
CREATE TABLE transaction_log (
  transaction_id BIGINT PRIMARY KEY AUTO_INCREMENT,
  transaction_type ENUM('PAYMENT', 'REFUND', 'INCOME', 'WITHDRAW') NOT NULL,
  from_user_id BIGINT,
  to_user_id BIGINT,
  amount DECIMAL(12, 2) NOT NULL,
  related_no VARCHAR(50),  -- 关联单号（订单号、合同号等）
  before_tenant_balance DECIMAL(12, 2),    -- 操作前租客余额
  before_landlord_balance DECIMAL(12, 2),  -- 操作前房东余额
  after_tenant_balance DECIMAL(12, 2),     -- 操作后租客余额
  after_landlord_balance DECIMAL(12, 2),   -- 操作后房东余额
  conservation_verified BOOLEAN DEFAULT TRUE,  -- 守恒验证结果
  conservation_sum DECIMAL(15, 2),         -- 守恒总额
  remark VARCHAR(255),
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (from_user_id) REFERENCES user(id),
  FOREIGN KEY (to_user_id) REFERENCES user(id),
  INDEX idx_type_created (transaction_type, created_at),
  INDEX idx_related_no (related_no)
);
```

## 🚨 错误处理代码示例

```java
@RestControllerAdvice
public class GlobalExceptionHandler {
  
  /**
   * 资金守恒异常
   */
  @ExceptionHandler(ConservationViolationException.class)
  public ResponseEntity<?> handleConservationViolation(
      ConservationViolationException e) {
    return ResponseEntity
        .status(409)
        .body(new ApiResponse(
            409,
            "资金守恒检查失败，操作已回滚: " + e.getMessage(),
            null
        ));
  }
  
  /**
   * 事务异常 - 数据库层面的异常
   */
  @ExceptionHandler(DataIntegrityViolationException.class)
  public ResponseEntity<?> handleDataIntegrity(
      DataIntegrityViolationException e) {
    return ResponseEntity
        .status(500)
        .body(new ApiResponse(
            500,
            "数据操作异常，事务已回滚",
            null
        ));
  }
  
  /**
   * 并发异常 - 行为在提交时被锁定
   */
  @ExceptionHandler(OptimisticLockingFailureException.class)
  public ResponseEntity<?> handleOptimisticLocking(
      OptimisticLockingFailureException e) {
    return ResponseEntity
        .status(409)
        .body(new ApiResponse(
            409,
            "数据已被其他用户修改，请重试",
            null
        ));
  }
}
```
