# Redis和分布式锁实现示例

## 🔐 分布式锁使用场景汇总

### 1. 用户服务 - 登录限流分布式锁
**场景**: 防止用户密码暴力破解
**文件**: `apartment-user-service/src/main/java/com/example/user/service/impl/UserServiceImpl.java`

```java
@Service
@RequiredArgsConstructor
public class UserServiceImpl implements UserService {
    
    private final RedissonClient redissonClient;
    private final UserMapper userMapper;
    
    @Override
    public String login(String username, String password) {
        // 登录限流锁: 每个用户每5秒只能尝试登录一次
        String lockKey = "user:login:limit:" + username;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            // 尝试获取锁,0秒等待,5秒自动释放
            boolean acquired = lock.tryLock(0, 5, TimeUnit.SECONDS);
            if (!acquired) {
                throw new RuntimeException("登录尝试过于频繁,请5秒后再试");
            }
            
            // 执行登录逻辑
            User user = userMapper.selectOne(new LambdaQueryWrapper<User>()
                    .eq(User::getUsername, username));
            
            if (user == null || !passwordEncoder.matches(password, user.getPassword())) {
                throw new RuntimeException("用户名或密码错误");
            }
            
            // 生成Token
            String token = jwtUtil.generateToken(user.getId(), user.getUsername());
            
            // 将Token缓存到Redis,有效期1小时
            RBucket<String> tokenBucket = redissonClient.getBucket("user:token:" + user.getId());
            tokenBucket.set(token, 1, TimeUnit.HOURS);
            
            return token;
        } catch (InterruptedException e) {
            throw new RuntimeException("获取锁失败");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
}
```

### 2. 订单服务 - 防重复下单分布式锁
**场景**: 同一用户对同一房源不能同时创建多个订单
**文件**: `apartment-order-service/src/main/java/com/example/order/service/impl/OrderServiceImpl.java`

```java
@Service
@RequiredArgsConstructor
public class OrderServiceImpl implements OrderService {
    
    private final RedissonClient redissonClient;
    private final OrderMapper orderMapper;
    
    @Override
    @Transactional(rollbackFor = Exception.class)
    public Long createOrder(CreateOrderDTO dto) {
        // 防重复下单锁: 用户ID + 房源ID
        String lockKey = "order:lock:" + dto.getUserId() + ":" + dto.getHouseId();
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            // 尝试获取锁,最多等待5秒,锁自动释放时间10秒
            boolean acquired = lock.tryLock(5, 10, TimeUnit.SECONDS);
            if (!acquired) {
                throw new RuntimeException("订单创建中,请勿重复提交");
            }
            
            // 检查是否已有未完成的订单
            LambdaQueryWrapper<Order> query = new LambdaQueryWrapper<>();
            query.eq(Order::getUserId, dto.getUserId())
                 .eq(Order::getHouseId, dto.getHouseId())
                 .in(Order::getStatus, "PENDING", "PAID");
            
            Long count = orderMapper.selectCount(query);
            if (count > 0) {
                throw new RuntimeException("您已有该房源的未完成订单");
            }
            
            // 生成订单号(使用Redis序列号)
            String orderNo = generateOrderNo();
            
            // 创建订单
            Order order = new Order();
            order.setOrderNo(orderNo);
            order.setUserId(dto.getUserId());
            order.setHouseId(dto.getHouseId());
            order.setLandlordId(dto.getLandlordId());
            order.setStartDate(dto.getStartDate());
            order.setEndDate(dto.getEndDate());
            order.setTotalAmount(dto.getTotalAmount());
            order.setStatus("PENDING");
            order.setCreatedAt(LocalDateTime.now());
            
            orderMapper.insert(order);
            
            // 缓存订单信息到Redis,30分钟
            RBucket<Order> orderCache = redissonClient.getBucket("order:info:" + order.getId());
            orderCache.set(order, 30, TimeUnit.MINUTES);
            
            return order.getId();
        } catch (InterruptedException e) {
            throw new RuntimeException("获取锁失败");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
    
    /**
     * 使用Redis生成唯一订单号
     */
    private String generateOrderNo() {
        RAtomicLong counter = redissonClient.getAtomicLong("order:no:seq");
        long seq = counter.incrementAndGet();
        
        // 格式: ORD + 日期 + 序列号
        String date = LocalDate.now().format(DateTimeFormatter.BASIC_ISO_DATE);
        return String.format("ORD%s%08d", date, seq);
    }
}
```

### 3. 支付服务 - 防重复支付分布式锁
**场景**: 确保订单不会被重复支付
**文件**: `apartment-payment-service/src/main/java/com/example/payment/service/impl/PaymentServiceImpl.java`

```java
@Service
@RequiredArgsConstructor
public class PaymentServiceImpl implements PaymentService {
    
    private final RedissonClient redissonClient;
    private final PaymentMapper paymentMapper;
    private final AccountService accountService;
    
    @Override
    @Transactional(rollbackFor = Exception.class)
    public Long pay(Long orderId, Long userId, BigDecimal amount, String paymentMethod) {
        // 防重复支付锁: 订单ID
        String lockKey = "payment:lock:" + orderId;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            // 尝试获取锁,最多等待5秒,锁自动释放时间30秒
            boolean acquired = lock.tryLock(5, 30, TimeUnit.SECONDS);
            if (!acquired) {
                throw new RuntimeException("支付处理中,请勿重复提交");
            }
            
            // 检查订单是否已支付
            LambdaQueryWrapper<Payment> query = new LambdaQueryWrapper<>();
            query.eq(Payment::getOrderId, orderId)
                 .eq(Payment::getStatus, "SUCCESS");
            
            Long count = paymentMapper.selectCount(query);
            if (count > 0) {
                throw new RuntimeException("订单已支付,请勿重复操作");
            }
            
            // 生成支付流水号
            String paymentNo = generatePaymentNo();
            
            // 创建支付记录
            Payment payment = new Payment();
            payment.setPaymentNo(paymentNo);
            payment.setOrderId(orderId);
            payment.setUserId(userId);
            payment.setAmount(amount);
            payment.setPaymentMethod(paymentMethod);
            payment.setStatus("PENDING");
            payment.setCreatedAt(LocalDateTime.now());
            
            paymentMapper.insert(payment);
            
            // 执行支付逻辑(扣款)
            if ("BALANCE".equals(paymentMethod)) {
                // 余额支付 - 需要加锁
                accountService.deduct(userId, amount);
            } else {
                // 其他支付方式...
            }
            
            // 更新支付状态
            payment.setStatus("SUCCESS");
            payment.setPaidAt(LocalDateTime.now());
            paymentMapper.updateById(payment);
            
            // 缓存支付状态到Redis
            RBucket<String> statusCache = redissonClient.getBucket("payment:status:" + payment.getId());
            statusCache.set("SUCCESS", 1, TimeUnit.HOURS);
            
            return payment.getId();
        } catch (InterruptedException e) {
            throw new RuntimeException("获取锁失败");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
    
    /**
     * 使用Redis生成唯一支付流水号
     */
    private String generatePaymentNo() {
        RAtomicLong counter = redissonClient.getAtomicLong("payment:no:seq");
        long seq = counter.incrementAndGet();
        
        // 格式: PAY + 日期 + 序列号
        String date = LocalDate.now().format(DateTimeFormatter.BASIC_ISO_DATE);
        return String.format("PAY%s%08d", date, seq);
    }
}
```

### 4. 支付服务 - 账户余额扣减分布式锁
**场景**: 防止并发扣款导致的超扣
**文件**: `apartment-payment-service/src/main/java/com/example/payment/service/impl/AccountServiceImpl.java`

```java
@Service
@RequiredArgsConstructor
public class AccountServiceImpl implements AccountService {
    
    private final RedissonClient redissonClient;
    private final AccountMapper accountMapper;
    
    @Override
    @Transactional(rollbackFor = Exception.class)
    public void deduct(Long userId, BigDecimal amount) {
        // 账户余额扣减锁: 用户ID
        String lockKey = "account:lock:" + userId;
        RLock lock = redissonClient.getLock(lockKey);
        
        try {
            // 尝试获取锁,最多等待5秒,锁自动释放时间10秒
            boolean acquired = lock.tryLock(5, 10, TimeUnit.SECONDS);
            if (!acquired) {
                throw new RuntimeException("账户处理中,请稍后再试");
            }
            
            // 查询账户余额
            Account account = accountMapper.selectById(userId);
            if (account == null) {
                throw new RuntimeException("账户不存在");
            }
            
            if (account.getBalance().compareTo(amount) < 0) {
                throw new RuntimeException("账户余额不足");
            }
            
            // 扣减余额
            BigDecimal newBalance = account.getBalance().subtract(amount);
            account.setBalance(newBalance);
            account.setUpdatedAt(LocalDateTime.now());
            
            accountMapper.updateById(account);
            
            // 更新Redis缓存中的余额
            RBucket<BigDecimal> balanceCache = redissonClient.getBucket("account:balance:" + userId);
            balanceCache.set(newBalance, 5, TimeUnit.MINUTES);
            
        } catch (InterruptedException e) {
            throw new RuntimeException("获取锁失败");
        } finally {
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
    
    /**
     * 查询账户余额(带缓存)
     */
    @Override
    public BigDecimal getBalance(Long userId) {
        // 先从Redis获取
        RBucket<BigDecimal> balanceCache = redissonClient.getBucket("account:balance:" + userId);
        BigDecimal balance = balanceCache.get();
        
        if (balance != null) {
            return balance;
        }
        
        // 缓存未命中,从数据库查询
        Account account = accountMapper.selectById(userId);
        if (account != null) {
            balance = account.getBalance();
            // 写入缓存,5分钟过期
            balanceCache.set(balance, 5, TimeUnit.MINUTES);
            return balance;
        }
        
        return BigDecimal.ZERO;
    }
}
```

## 📊 分布式锁场景汇总表

| 服务 | 场景 | 锁键格式 | 等待时间 | 超时时间 | 目的 |
|------|------|---------|----------|----------|------|
| 用户服务 | 登录限流 | `user:login:limit:{username}` | 0秒 | 5秒 | 防止暴力破解 |
| 订单服务 | 防重复下单 | `order:lock:{userId}:{houseId}` | 5秒 | 10秒 | 防止重复提交 |
| 支付服务 | 防重复支付 | `payment:lock:{orderId}` | 5秒 | 30秒 | 防止重复支付 |
| 支付服务 | 账户余额扣减 | `account:lock:{userId}` | 5秒 | 10秒 | 防止超扣 |
